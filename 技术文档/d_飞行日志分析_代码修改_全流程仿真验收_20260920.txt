三、代码问题与原因判断
1. 旧版 PID 用 ey=(CY-cy)-FY*tan(pitch)，桶反投影使用另一符号关系；横滚只进部分路径。在有姿态时，同一检测框的像素残差、反投影桶点和期望像素可能互相不一致。所谓“下降到 1.2m 切纯视觉”并没有断开桶点的反投影：DROP 仍用它做身份锁与位置目标，再用像素残差积分修正，故两者仍能冲突。第一桶较大且波动的误差与此问题相容，但无法单凭日志定量归因。
2. 第一瓶触发时旧代码立即将 _drop_count 加一。后续 DROP 循环可能按第二瓶的横向补偿计算，B1 与 B2 配置差 0.15m。再加上收敛检查的提前 return 挡在舵机复位计时之前，第一瓶触发后输出保持及目标都可能被后续图像变化扰动。日志中第一瓶触发后约 11 秒才离开 DROP，与此路径相符。
3. 旧版缓存像素误差可在检测中断时沿用，检测框不变还可能被当作停更；放行判断使用被死区置零的误差。这会高估“已经稳定收敛”的可信度。
4. 该程序仍依赖假定的固定下视相机、FOV 推算内参和以起飞平面为参考的高度。当前资料不足以验证相机朝向、镜像/旋转、相机至桶口真实距离以及释放机构动作延时。对“右后偏”不能直接写死一个左前补偿量。

def camera_pixel_to_level(cx, cy, pitch, roll):
    """下视像素射线 -> 水平航向坐标中的 (前向/H, 右向/H)。

    固定安装假设：图像上=机头，图像右=机体右，相机光轴=机体下。
    PX4 FRD -> NED 使用 Rz(yaw) Ry(pitch) Rx(roll)。这里去除
    roll/pitch，yaw 由 _body_to_ned 处理。视觉误差和反投影共用本模型。
    """
    if not all(math.isfinite(v) for v in (cx, cy, pitch, roll)):
        return None
    fwd = (CY - cy) / FY
    right = (cx - CX) / FX
    cr, sr = math.cos(roll), math.sin(roll)
    cp, sp = math.cos(pitch), math.sin(pitch)
    rolled_right = cr * right - sr
    rolled_down = sr * right + cr
    level_fwd = cp * fwd + sp * rolled_down
    level_down = -sp * fwd + cp * rolled_down
    if level_down <= 1e-6:  # 射线平行于地面或指向天空
        return None
    return level_fwd / level_down, rolled_right / level_down


def level_to_camera_pixel(fwd, right, height, pitch, roll):
    """camera_pixel_to_level 的逆变换，返回像素 (cx, cy) 或 None。"""
    if not all(math.isfinite(v) for v in (fwd, right, height, pitch, roll)) or height <= 0.0:
        return None
    cr, sr = math.cos(roll), math.sin(roll)
    cp, sp = math.cos(pitch), math.sin(pitch)
    body_fwd = cp * fwd - sp * height
    unpitched_down = sp * fwd + cp * height
    body_right = cr * right + sr * unpitched_down
    body_down = -sr * right + cr * unpitched_down
    if body_down <= 1e-6:
        return None
    return CX + FX * body_right / body_down, CY - FY * body_fwd / body_down


class PIDController:
    def __init__(self, kp, ki, kd, max_vel, int_limit, dead_zone_px=0.0):
        self.kp, self.ki, self.kd = kp, ki, kd
        self.max_vel, self.int_limit = max_vel, int_limit
        self.dead_zone_px = dead_zone_px
        self.prev_ex = self.prev_ey = 0.0
        self.int_ex = self.int_ey = 0.0

    def reset(self):
        self.prev_ex = self.prev_ey = 0.0
        self.int_ex = self.int_ey = 0.0

    def compute(self, roi, dt, pixel_offset_x=0.0, pitch=0.0, roll=0.0, height_scale=1.0):
        """纯 PD 像素偏差→机体速度.

        返回: (v_fwd, v_lat, err_px, ex, ey)
          v_fwd: 机体前向速度 (m/s), 正值=前进
          v_lat: 机体右向速度 (m/s), 正值=右移
          err_px: 按水平地面距离折算的等效像素误差，保留控制死区内的真实残差
          ex, ey: 等效横向/纵向误差 (ey>0=目标在前方, 需前进)

        height_scale = H / CRUISE_HEIGHT: 低空自动缩小 KP/KD, 消除像素/物理比变化.
        """
        cx = roi.x_offset + roi.width / 2.0
        cy = roi.y_offset + roi.height / 2.0
        projected = camera_pixel_to_level(cx, cy, pitch, roll)
        if projected is None:
            self.reset()
            return 0.0, 0.0, float('inf'), float('inf'), float('inf')
        fwd_per_h, right_per_h = projected
        ex = right_per_h * FX - pixel_offset_x
        ey = fwd_per_h * FY
        raw_ex, raw_ey = ex, ey  # 死区只影响控制输出，释放判据保留未抹零误差

        if abs(ex) < self.dead_zone_px: ex = 0.0
        if abs(ey) < self.dead_zone_px: ey = 0.0

        dt = max(dt, 0.001)
        self.int_ex += ex * dt
        self.int_ey += ey * dt
        self.int_ex = max(-self.int_limit, min(self.int_limit, self.int_ex))
        self.int_ey = max(-self.int_limit, min(self.int_limit, self.int_ey))

        dex = (ex - self.prev_ex) / dt
        dey = (ey - self.prev_ey) / dt

        kp = self.kp * height_scale
        kd = self.kd * height_scale

        v_lat = kp * ex + self.ki * self.int_ex + kd * dex
        v_fwd = kp * ey + self.ki * self.int_ey + kd * dey
        v_lat = max(-self.max_vel, min(self.max_vel, v_lat))
        v_fwd = max(-self.max_vel, min(self.max_vel, v_fwd))

        self.prev_ex, self.prev_ey = ex, ey
        return v_fwd, v_lat, math.hypot(raw_ex, raw_ey), raw_ex, raw_ey

def geo_lock_decide(est_x, est_y, anchor_x, anchor_y, gate, weight,
                    prev_est, reject_n, reanchor_n, reanchor_rad, reanchor_max):
    """反投影估计 vs 锚点 → 目标位置. 返回 (mode, tx, ty, reject_n, prev_est).

    accept   : 距锚点 < gate → 采信 (weight*反投影 + (1-weight)*锚点)
    reanchor : 连续 reanchor_n 帧一致地超 gate 且 < reanchor_max → 认可新位置
               (同一个桶因高度变化引起的反投影漂移, 不是误检跳变)
    reject   : 超 gate 但只出现 1~reanchor_n-1 帧 → 暂用锚点, 记账等待一致性确认
    block    : 超 reanchor_max → 疑似别的桶, 恒用锚点且永不重锚
    """
    dist = math.hypot(est_x - anchor_x, est_y - anchor_y)
    if dist < gate:
        return ('accept', weight * est_x + (1.0 - weight) * anchor_x,
                weight * est_y + (1.0 - weight) * anchor_y, 0, None)
    if dist > reanchor_max:
        return ('block', anchor_x, anchor_y, 0, None)
    if prev_est is not None and math.hypot(est_x - prev_est[0], est_y - prev_est[1]) < reanchor_rad:
        n = reject_n + 1
    else:
        n = 1
    if n >= reanchor_n:
        return ('reanchor', est_x, est_y, 0, None)
    return ('reject', anchor_x, anchor_y, n, (est_x, est_y))

    def _cb_local(self, m):
        self.x, self.y, self.z = m.x, m.y, m.z
        if math.isfinite(m.vx): self.vx = m.vx
        if math.isfinite(m.vy): self.vy = m.vy
        if math.isfinite(m.vz): self.vz = m.vz
        if math.isfinite(m.heading):
            self.heading = m.heading
        # 记录起飞参考点 (航向用硬编码 HEADING_DEG, 不再锁机头)
        # 实机无 xy_valid (室内无 GPS), 有 heading 即可记录参考点

            self._takeoff_x, self._takeoff_y, self._takeoff_z = self.x, self.y, self.z
            self.get_logger().info(
                f"Takeoff ref: NED({self._takeoff_x:.2f},{self._takeoff_y:.2f},{self._takeoff_z:.2f}) "
                f"HEADING_DEG={HEADING_DEG:.0f}deg (hardcoded)")

    def _cb_attitude(self, m):
        w, x, y_, z_ = float(m.q[0]), float(m.q[1]), float(m.q[2]), float(m.q[3])
        self._roll  = math.atan2(2.0*(w*x + y_*z_), 1.0 - 2.0*(x*x + y_*y_))
        self._pitch = math.asin(max(-1.0, min(1.0, 2.0*(w*y_ - z_*x))))

    def _cb_detect(self, msg):
        """接收 JSON 多桶检测.
        HOVER_SEARCH (静止): ROI→NED 估算桶坐标 → 锁定.
        TRACK/DESCEND (移动): 已知坐标→期望像素投影, 像素空间最近邻匹配.
        """
        try:
            data = json.loads(msg.data)
        except Exception:
            return
        detections = data.get("buckets", [])

        valid = []
        for d in detections:
            roi = RegionOfInterest()
            roi.x_offset = int(d["x"]); roi.y_offset = int(d["y"])
            roi.width = int(d["w"]); roi.height = int(d["h"])
            roi.do_rectify = False
            est = self._roi_to_ned(roi)
            if est is not None:
                cx = roi.x_offset + roi.width / 2.0
                cy = roi.y_offset + roi.height / 2.0
                valid.append((est[0], est[1], roi, d, cx, cy))

        # 2026-09-12: 同帧重复框合并 (防同一个桶被喂出两个追踪器 → 两个目标其实是同一个桶)
        valid = self._dedup_dets(valid)
        self._bucket_ests = valid
        self._update_trackers(valid)  # v9: 每帧更新追踪器
        if not valid:
            return

        if self._current_target_ned is not None:
            # TRACK/DESCEND/DROP: 几何身份锁 — 找 NED 估计最接近目标坐标的 ROI
            # (不用 nearest-center, 防止多桶时追错目标)
            best_roi = None; best_d = None
            best_geo_dist = float('inf')
            GEO_IDENTITY_MAX = 1.5  # 几何身份门限 (m), v6: 2.5->1.5 < 最小桶间距 2.2m
            for bx, by, roi, d, cx, cy in valid:
                geo_dist = math.hypot(bx - self._current_target_ned[0],
                                      by - self._current_target_ned[1])
                if geo_dist < best_geo_dist:
                    best_geo_dist = geo_dist
                    best_roi = roi; best_d = d
            if best_roi is not None and best_geo_dist < GEO_IDENTITY_MAX:
                roi_key = (best_d["x"], best_d["y"], best_d["w"], best_d["h"])
                self._update_fresh(roi_key)
                self.latest_roi = best_roi
                self._last_matched_detect_time = self.get_clock().now()
            # 找不到匹配: 不更新 latest_roi, 触发丢检保护
        else:
            # HOVER_SEARCH: 取离图像中心最近的检测, 确保选到正下方的桶
            best = min(valid, key=lambda v: math.hypot(v[4] - CX, v[5] - CY))
            _, _, roi, d, _, _ = best
            roi_key = (d["x"], d["y"], d["w"], d["h"])
            self._update_fresh(roi_key)
            self.latest_roi = roi
            self._last_matched_detect_time = self.get_clock().now()

    def _update_fresh(self, roi_key):
        """更新检测新鲜度 (内联去重逻辑)."""
        if roi_key == self._last_roi_key:
            self._roi_dup_count += 1
            # 视觉伺服时帧间移动 <1px 很常见, 提高阈值防误判
            if self._roi_dup_count < 25:
                self._last_detect_time = self.get_clock().now()
        else:
            self._roi_dup_count = 0
            self._last_roi_key = roi_key
            self._last_detect_time = self.get_clock().now()

    def _confirm_detection_stable(self):
        if not self._detect_fresh(): self._confirm_start_time = None; return False
        roi = self.latest_roi
        cx = roi.x_offset + roi.width / 2.0; cy = roi.y_offset + roi.height / 2.0
        if self._confirm_start_time is None:
            self._confirm_start_time = self.get_clock().now()
            self._confirm_first_cx = cx; self._confirm_first_cy = cy; return False
        drift = math.hypot(cx - self._confirm_first_cx, cy - self._confirm_first_cy)
        if drift > TRACK_ENTRY_ROI_DRIFT: self._confirm_start_time = None; return False
        return (self.get_clock().now() - self._confirm_start_time).nanoseconds * 1e-9 >= TRACK_ENTRY_CONFIRM_TIME

    # ========== 多桶追踪器 (v9) ==========
    def _dedup_dets(self, ests):
        """2026-09-12: 同帧内重复检测框合并.

        动机 (d 2026-09-12 仿真): HOVER_SEARCH 期间 3 个桶检测出 5 个框 (DETx5), 多出的框
        经 _roi_to_ned 得到不同 NED → 追踪器把同一个桶记成两个 tracker, 各自通过 1.5s 确认
        → 选桶选出两个"影子桶" (实测两目标间距 1.78m, 而场上最小真实桶间距 2.24m)
        → 两次投弹落在同一个桶 → 有效得分 10/20. 实机 YOLO 低置信度重复框同理.

        判据: 同帧内两框的 NED 估计相距 < DET_DUP_MERGE_DIST 即视为同一个桶, 保留检测框较大者.
        安全性: 同一帧/同一机位下, 同一个桶的两个框系统性偏差完全相同 (差异仅几 cm~0.3m);
        而最小真实桶间距 2.24m 即使各带 ±0.4m 估计噪声仍有 1.44m > 0.8m, 不会误合并真实桶.
        """
        if len(ests) <= 1:
            return ests
        kept = []
        for e in sorted(ests, key=lambda v: -(v[2].width * v[2].height)):
            if all(math.hypot(e[0] - k[0], e[1] - k[1]) >= DET_DUP_MERGE_DIST for k in kept):
                kept.append(e)
        n_drop = len(ests) - len(kept)
        if n_drop != getattr(self, '_dedup_last_n', 0):
            self._dedup_last_n = n_drop
            self.get_logger().info(
                f"  同帧重复框合并: {len(ests)} -> {len(kept)} "
                f"(相距 < {DET_DUP_MERGE_DIST}m 视为同一个桶)")
        return kept

    def _update_trackers(self, bucket_ests):
        """v9 多桶追踪器: 每帧 NED 最近邻匹配, 每桶独立时间确认 ≥1.5s.

        每个物理桶对应一个追踪器 {ned, first_seen, last_seen, confirmed, roi_center}.
        匹配: 当前帧各检测的 NED 估计与已有追踪器做最近邻匹配 (<1.5m),
              匹配成功 → 指数平滑更新位置 (alpha=0.3), 刷新时间, 检查确认.
              未匹配 → 为新检测创建新追踪器.
        清理: 超时 >2s 未更新且本轮未匹配 → 丢弃.

        防御 YOLO 瞬时误检: 单个检测必须持续 ≥1.5s 才被确认(confirmed=True),
        瞬时的背景干扰检测不会存活足够久.
        """
        now = self.get_clock().now()
        now_s = now.nanoseconds * 1e-9
        matched = set()

        for bx, by, roi, d, cx, cy in bucket_ests:
            best_t, best_d = None, float('inf')
            for t in self._trackers:
                dist = math.hypot(bx - t["ned"][0], by - t["ned"][1])
                if dist < best_d:
                    best_d = dist
                    best_t = t
            if best_t is not None and best_d < TRACKER_NED_MATCH:
                alpha = 0.3
                old_x, old_y = best_t["ned"]
                best_t["ned"] = (old_x + alpha * (bx - old_x),
                                 old_y + alpha * (by - old_y))
                best_t["last_seen"] = now_s
                best_t["roi_center"] = (cx, cy)
                best_t["area"] = float(roi.width * roi.height)
                if not best_t["confirmed"] and (now_s - best_t["first_seen"]) >= TRACKER_CONFIRM_TIME:
                    best_t["confirmed"] = True
                    self.get_logger().info(
                        f"Tracker confirmed: NED({best_t['ned'][0]:.1f},{best_t['ned'][1]:.1f}) "
                        f"after {now_s - best_t['first_seen']:.1f}s")
                matched.add(id(best_t))
            else:
                self._trackers.append({
                    "ned": (bx, by),
                    "first_seen": now_s,
                    "last_seen": now_s,
                    "confirmed": False,
                    "roi_center": (cx, cy),
                    "area": float(roi.width * roi.height),
                })

        self._trackers = [t for t in self._trackers
                          if (now_s - t["last_seen"] < TRACKER_TIMEOUT)
                          or id(t) in matched]

    def _enter_step(self, step):
        prev_step = self.step
        if prev_step == STEP_DROP and step != STEP_DROP:
            self._aux5_value = AUX5_LOCK
        prev_elapsed = self._step_elapsed()
        self.step = step; self._step_start_time = self.get_clock().now()
        self._stabilize_start = None
        self._detect_loss_logged = False; self._detect_loss_start = None
        self._confirm_start_time = None; self._stabilize_lost_frames = 0
        self._drop_triggered = False; self._drop_signal_start = None
        self._drop_hold_target = None
        self._drop_converge_start = None  # 2026-08-31: 收敛即投计时
        # 2026-09-13 实机投弹精度修复: 身份锁一致性记账 + 像素残差积分 (DROP 闭环)
        self._geo_reject_n = 0          # 连续被拒帧数
        self._geo_pending_est = None    # 上一帧被拒的反投影估计
        self._geo_mode = None           # 上一帧身份锁模式 (模式变化时打日志)
        self._servo_bias_x = 0.0        # 像素残差积分项 (抵消反投影系统性偏置)
        self._servo_bias_y = 0.0
        self._bias_sat_logged = False   # 积分项饱和只告警一次
        if step == STEP_POSITION_REFINE:
            self._refine_ests = []
            self._roi_dup_count = 0; self._last_roi_key = None
            self._last_detect_time = self.get_clock().now()
        if step == STEP_VISUAL_SERVO:
            self.pid.reset()
            self._servo_active = True
            self._servo_target_x = self._current_target_ned[0]
            self._servo_target_y = self._current_target_ned[1]
            self._servo_converge_start = None
            self._decimation_ctr = 0
        if step in (STEP_PRE_SERVO, STEP_VISUAL_SERVO):
            self._desc_entry_z = self.z  # 下降时间锚定起点 (2026-08-15 d)
        if step == STEP_DROP:
            self._decimation_ctr = 0
            self._last_drop_err = None
            # 2026-09-12: 收敛即投计时复位 (与 e 对齐). 原实现只在 __init__ 初始化一次,
            # 若第 1 桶是"收敛即投"结束的, _drop_converge_start 会残留到第 2 桶 →
            # 第 2 桶进 DROP 后第一帧只要"等效地面距离<门限"且已到高度, sim_t-旧时间戳(≈100s)
            # 就 > DROP_CONVERGE_EARLY → "持续 1.5s" 被瞬间满足 = 零驻留提前投弹.
            self._drop_converge_start = None
            # 2026-09-13: 身份锁记账 + 像素残差积分 每桶复位
            self._geo_reject_n = 0
            self._geo_pending_est = None
            self._geo_mode = None
            self._servo_bias_x = 0.0
            self._servo_bias_y = 0.0
            self._bias_sat_logged = False
        if step == STEP_FORWARD:
            self._fwd_exit_frames = 0
            self._fwd_arrive_frames = 0
        if step == STEP_TAKEOFF and self._mission_start_time is None:
            self._mission_start_time = self.get_clock().now()
        if step == STEP_STABILIZE: self._stabilize_yaw = None
        if step == STEP_HOVER_SEARCH:
            self._search_good_frames = 0
            self._trackers = []  # v9: 新搜索, 清空旧追踪器
        if step == STEP_RETURN:
            self._current_target_ned = None
            self._target_img_idx = -1
            self._target_buckets = []
            self._return_sub = 0
            self._return_sub_start = None
            self._stab_pos_hist.clear()
        if step == STEP_LAND:
            self._land_entry_z = None
        if step == STEP_CLIMB:
            self._climb_done = False
        if step == STEP_SCOUT:
            self._scout_hover_start = None
            self._scout_sub = 0
            self._scout_sub_start = None
        # 阶段切换日志 (2026-08-15 d): 每阶段进入位置/高度 + 前一阶段耗时
        if self.x is not None and self.home_z is not None:
            if prev_step == step:
                self.get_logger().info(f"STEP {STEP_NAME[step]} re-entered")
            else:
                self.get_logger().info(
                    f"STEP {STEP_NAME[prev_step]}->{STEP_NAME[step]} "
                    f"(prev {prev_elapsed:.1f}s) NED({self.x:.2f},{self.y:.2f},{self.z:.2f}) "
                    f"H={self.home_z-self.z:.2f}m")

    def _step_elapsed(self):
        if self._step_start_time is None: return 0.0
        return (self.get_clock().now() - self._step_start_time).nanoseconds * 1e-9

    def _detect_fresh(self):
        if self.latest_roi is None: return False
        return (self.get_clock().now() - self._last_detect_time).nanoseconds * 1e-9 < DETECT_TIMEOUT

    def _drop_detect_age(self):
        if self.latest_roi is None or self._last_matched_detect_time is None:
            return float('inf')
        return (self.get_clock().now() - self._last_matched_detect_time).nanoseconds * 1e-9

    def _drop_detection_fresh(self):
        return 0.0 <= self._drop_detect_age() <= DROP_DETECT_MAX_AGE

    # ========== 几何工具 ==========
    def _roi_to_ned(self, roi):
        """像素 → 相对home的NED偏移 (非PX4绝对坐标).

        与 PID 共用 camera_pixel_to_level；H 是起飞参考平面相对高度。
        前/右距离经航向旋转到 NED；无有效地面交点时返回 None。
        """
        if self.heading is None or self.home_z is None or self.home_x is None: return None
        H = self.home_z - self.z
        if not math.isfinite(H) or H < 0.3: return None
        cx = roi.x_offset + roi.width / 2.0; cy = roi.y_offset + roi.height / 2.0
        projected = camera_pixel_to_level(cx, cy, self._pitch, self._roll)
        if projected is None: return None
        cam_fwd, cam_right = (v * H for v in projected)
        dx, dy, _ = self._body_to_ned(cam_fwd, cam_right)
        return (self.x - self.home_x + dx, self.y - self.home_y + dy)

    def _body_to_ned(self, v_fwd, v_lat, v_down=0.0):
        cos_h, sin_h = math.cos(self.heading), math.sin(self.heading)
        return (v_fwd * cos_h - v_lat * sin_h,
                v_fwd * sin_h + v_lat * cos_h, v_down)

    def _ned_to_body(self, dx, dy):
        """NED 世界偏移 → 机体坐标系 (fwd, right)."""
        cos_h, sin_h = math.cos(self.heading), math.sin(self.heading)
        fwd = dx * cos_h + dy * sin_h
        lat = -dx * sin_h + dy * cos_h
        return fwd, lat

    def _ned_to_expected_pixel(self, target_ned):
        """已知桶 NED → 期望像素坐标 (用于 TRACK 阶段身份锁定)."""
        if self.heading is None or self.home_z is None or self.home_x is None:
            return None
        H = self.home_z - self.z
        if H < 0.3:
            return None
        dx = (self.home_x + target_ned[0]) - self.x
        dy = (self.home_y + target_ned[1]) - self.y
        cam_fwd, cam_right = self._ned_to_body(dx, dy)
        return level_to_camera_pixel(cam_fwd, cam_right, H, self._pitch, self._roll)

    def _finalize_refine(self):
        """平均收集到的 ROI→NED 估计, 更新 _current_target_ned.
        如果修正量 > POSITION_REFINE_MAX_CORRECTION 则拒绝 (防止身份混淆).
        """
        ests = getattr(self, '_refine_ests', [])
        if len(ests) >= POSITION_REFINE_MIN_SAMPLES:
            avg_x = sum(e[0] for e in ests) / len(ests)
            avg_y = sum(e[1] for e in ests) / len(ests)
            old_x, old_y = self._current_target_ned
            correction = math.hypot(avg_x - old_x, avg_y - old_y)
            if correction > POSITION_REFINE_MAX_CORRECTION:
                self.get_logger().warn(
                    f"Position refine REJECTED: correction {correction:.2f}m > "
                    f"{POSITION_REFINE_MAX_CORRECTION}m (identity confusion)")
            else:
                self._current_target_ned = (avg_x, avg_y)
                self.get_logger().info(
                    f"Position refined: ({old_x:.2f},{old_y:.2f}) -> ({avg_x:.2f},{avg_y:.2f}) "
                    f"delta=({avg_x-old_x:.3f},{avg_y-old_y:.3f})m n={len(ests)}")
        else:
            self.get_logger().info(
                f"Position refine skipped: only {len(ests)} samples (need {POSITION_REFINE_MIN_SAMPLES})")
        self._refine_ests = []

    def _bucket_target(self):
        """FORWARD 目标点, 相对 home 的 NED 偏移."""
        if self._heading0 is None: return (0.0, 0.0)
        cos_h0, sin_h0 = math.cos(self._heading0), math.sin(self._heading0)
        return (FORWARD_DIST * cos_h0,
                FORWARD_DIST * sin_h0)

    def _scout_target(self):
        """侦察区中心, 相对 home 的 NED 偏移."""
        if self._heading0 is None: return (0.0, 0.0)
        cos_h0, sin_h0 = math.cos(self._heading0), math.sin(self._heading0)
        return (SCOUT_DIST * cos_h0,
                SCOUT_DIST * sin_h0)

    def _scout_edge_target(self):
        """侦察区第一根线(近边)中点, 相对 home 的 NED 偏移. 2026-08-15: 直飞目标改近边防过冲."""
        if self._heading0 is None: return (0.0, 0.0)
        cos_h0, sin_h0 = math.cos(self._heading0), math.sin(self._heading0)
        return (SCOUT_EDGE_DIST * cos_h0,
                SCOUT_EDGE_DIST * sin_h0)

    def _fly_toward(self, tx, ty, speed, vz=0.0):
        """速度控制飞向绝对NED坐标 (tx, ty)。
        tx, ty 是绝对 NED 坐标 (不是相对 home 偏移)。
        调用方: SCOUT 传 home_x+sx, RETURN 传 home_x, home_y.
        """
        dist = math.hypot(tx - self.x, ty - self.y)
        if dist > 0.1:
            # 距离越近速度越低, dist<3m 时线性 ramp-down, 最低 FORWARD_MIN_SPEED
            ratio = max(FORWARD_MIN_SPEED / CRUISE_SPEED, min(1.0, dist / 3.0))
            vx = (tx - self.x) / dist * speed * ratio
            vy = (ty - self.y) / dist * speed * ratio
        else: vx = vy = 0.0
        return vx, vy, dist

    # 2026-09-11 按用户指示移除 _enforce_boundaries (软墙机制):
    # 实机事故 — 目标桶侧向 4.21m 超出 ±4m 软墙, 软墙每 1-2s 接管并跳过状态机
    # (主循环 `if self._enforce_boundaries(): return`), 视觉伺服被反复打断
    # (err 74→218px 发散), 任务中止。不再设置任何飞行边界。

    def _handle_detect_loss(self, target_z):
        if not self._detect_loss_logged:
            self._detect_loss_logged = True
            self.get_logger().warn(f"Detection lost in {STEP_NAME[self.step]}, hovering...")
        self._pub_vel(0.0, 0.0, (target_z - self.z) * TRACK_VZ_KP, self.heading)

    def _shutdown(self, reason="unknown"):
        self.timer.cancel()
        self.manual_timer.cancel()
        self._print_summary(reason)
        self.get_logger().info("Mission complete, shutting down")
        self.destroy_node()

    def _print_summary(self, reason):
        """任务总结报告 (2026-08-27): 结束原因/投弹明细/侦察/返航/着陆/总时长."""
        self.get_logger().info("=" * 60)
        self.get_logger().info("MISSION SUMMARY")
        self.get_logger().info(f"  End reason: {reason}")
        now = self.get_clock().now()
        total = None
        if self._mission_start_time is not None:
            total = (now - self._mission_start_time).nanoseconds * 1e-9
            self.get_logger().info(f"  Total time: {total:.1f}s")
        self.get_logger().info(
            f"  Buckets dropped: {len(self._drop_records)}/{getattr(self, '_target_count', 0)}")
        for r in self._drop_records:
            aim_str = f"NED({r['aim'][0]:.2f},{r['aim'][1]:.2f})" if r["aim"] else "unknown"
            pos = r["pos"]
            dev = math.hypot(pos[0] - r["aim"][0], pos[1] - r["aim"][1]) if r["aim"] else float("nan")
            err_str = ""
            if r["last_err"] is not None:
                ep, exl, eyl, hh = r["last_err"]
                err_str = f" last_err={ep:.0f}px(ex={exl:.0f},ey={eyl:.0f})"
            self.get_logger().info(
                f"  Drop #{r['idx']}: aim {aim_str} at NED({pos[0]:.3f},{pos[1]:.3f}) "
                f"H={r['H']:.2f}m vehicle_to_anchor={dev:.3f}m{err_str} actual_impact=unmeasured")
        if self._scout_done and self._scout_elapsed is not None:
            self.get_logger().info(f"  Scout: completed ({self._scout_elapsed:.1f}s, 7 sub-phases)")
        elif self.step == STEP_SCOUT:
            self.get_logger().info(f"  Scout: incomplete (aborted at sub={self._scout_sub})")
        else:
            self.get_logger().info("  Scout: not reached")
        if self._return_stab_info is not None:
            stab_t, drift = self._return_stab_info
            self.get_logger().info(f"  Return: stabilized in {stab_t:.1f}s (drift {drift:.2f}m)")
        else:
            self.get_logger().info("  Return: no stabilization (forced land)")
        if self._land_pos is not None:
            dist = math.hypot(self._land_pos[0] - self.home_x, self._land_pos[1] - self.home_y)
            self.get_logger().info(
                f"  Landing: NED({self._land_pos[0]:.2f},{self._land_pos[1]:.2f}) "
                f"dist_from_home={dist:.2f}m")
        else:
            self.get_logger().info("  Landing: not reached")
        if self._search_failed:
            self.get_logger().info("  NOTE: no valid targets in HOVER_SEARCH, skipped drops")
        if self._mission_timeout_forced:
            self.get_logger().info("  NOTE: mission timeout forced landing")
        self.get_logger().info("=" * 60)

        msg.valid = True; msg.data_source = ManualControlSetpoint.SOURCE_RC

        self._aux5_pwm_ctr += 1
        if self._aux5_pwm_ctr >= 20:
            self._aux5_pwm_ctr = 0
            self.aux5_pwm_pub.publish(Float32(data=float(1500.0 + self._aux5_value * 600.0)))

    # ================================================================
    # 主循环
    # ================================================================
    def control_loop(self):
        # v2: 用 _takeoff_x 做入口守卫 (home 延后到 STABILIZE 才锁定)
        if self._takeoff_x is None or self.heading is None: return

        sim_t = (self.get_clock().now() - self._start_wall).nanoseconds * 1e-9
        self._print_ctr += 1
        if self._print_ctr >= 20:
            tag = STEP_NAME.get(self.step, "???"); hdg = math.degrees(self.heading) if self.heading else 0
            has_det = f"DETx{len(self._bucket_ests)}" if self._bucket_ests else "no-det"; extra = ""
            if self.step in (STEP_TRACK, STEP_VISUAL_SERVO, STEP_DROP) and self._bucket_idx < len(self._target_buckets):
                extra = f" B{self._bucket_idx+1}/{len(self._target_buckets)}"
            if self.step == STEP_VISUAL_SERVO and self._servo_active:
                extra += " SERVO"
            if self.step == STEP_POSITION_REFINE:
                n = len(getattr(self, '_refine_ests', []))
                extra = f" refine_samples={n} elapsed={self._step_elapsed():.1f}s"
            if self.step == STEP_STABILIZE:
                extra = f" stabilizing {self._step_elapsed():.1f}s/{STABILIZE_HOVER_TIME}s"
            elif self.step == STEP_FORWARD and self.home_x is not None:
                tx, ty = self._bucket_target()
                # 2026-09-11 修复: tx/ty 是相对 home 偏移, 必须与相对位置相减.
                # 原写法混用绝对 self.x/self.y → 显示值虚高 |home| (实机日志假值 72.9m). 仅日志, 不影响控制.
                extra = f" dist_to_target={math.hypot((self.x-self.home_x)-tx,(self.y-self.home_y)-ty):.1f}m"
            elif self.step == STEP_HOVER_SEARCH:
                conf = sum(1 for t in self._trackers if t["confirmed"])
                extra = f" search {self._step_elapsed():.1f}s confirmed={conf}/{len(self._trackers)}"
            elif self.step == STEP_SCOUT:
                extra = f" scout sub={self._scout_sub}"
            elif self.step == STEP_RETURN and self.home_x is not None:
                dist_h = math.hypot(self.x-self.home_x, self.y-self.home_y)
                if self._return_sub == 0:
                    extra = f" ->home dist={dist_h:.1f}m"
                else:
                    drift = 0.0
                    if self._stab_pos_hist:
                        xs = [p[0] for p in self._stab_pos_hist]; ys = [p[1] for p in self._stab_pos_hist]
                        drift = max(max(xs)-min(xs), max(ys)-min(ys))
                    extra = f" stabilizing dist={dist_h:.2f}m drift={drift:.2f}m"
            self.get_logger().info(f"[{tag}] NED({self.x:.2f},{self.y:.2f},{self.z:.2f}) h={hdg:.0f}deg {has_det}{extra}")
            self._print_ctr = 0

        # 初始等待 (DDS 就绪)
        if not self._initial_wait_done:
            if self._init_wait_start is None: self._init_wait_start = self.get_clock().now()
            self._pub_vel(0.0, 0.0, 0.0, self.heading)
            elapsed = (self.get_clock().now() - self._init_wait_start).nanoseconds * 1e-9
            dds_ready = self._control_mode is not None
            if dds_ready and elapsed > 3.0: self._initial_wait_done = True
            elif elapsed > 30.0:
                self.get_logger().error("DDS wait timeout (30s): no PX4 topics")
                self._shutdown("dds_timeout"); return
            return

        # ARM / OFFBOARD (实机: VehicleControlMode 单源, 不用 VehicleStatus fallback)
        cm = self._control_mode
        armed = cm.flag_armed if cm is not None else False
        offboard = cm.flag_control_offboard_enabled if cm is not None else False
        if cm is None: self._pub_vel(0, 0, 0, self.heading); return
        if self._mode_req_start is None: self._mode_req_start = self.get_clock().now()
        elapsed = (self.get_clock().now() - self._mode_req_start).nanoseconds * 1e-9
        ref_x, ref_y = self._takeoff_x, self._takeoff_y
        ref_z = self._takeoff_z - CRUISE_HEIGHT
        yaw_live = self.heading if self.heading else 0.0
        if self.step == STEP_LAND and self._land_cmd_sent: pass
        elif armed and offboard:
            if not self._arm_offboard_logged:
                self._arm_offboard_logged = True
                self.get_logger().info(f"ARM + OFFBOARD active (t={elapsed:.1f}s), entering mission")
        elif elapsed > 30.0: self.get_logger().error("Mode timeout"); self._shutdown("mode_timeout"); return
        elif not armed:
            if not self._arm_req:
                self.get_logger().info("Requesting ARM...")
            if not self._arm_req or elapsed - self._last_arm_attempt > 1.0:
                self._arm_req = True; self._last_arm_attempt = elapsed; send_arm(self.cmd_pub, self)
            self._pub_pos(ref_x, ref_y, self._takeoff_z, yaw_live); return
        elif not offboard:
            if armed and not self._arm_logged:
                self._arm_logged = True
                self.get_logger().info("ARM OK, requesting OFFBOARD mode...")
            if not self._mode_req:
                self._mode_req = True
                self.get_logger().info("Requesting OFFBOARD mode...")
                self._send_cmd(176, param1=1.0, param2=6.0)
            self._pub_pos(ref_x, ref_y, self._takeoff_z, yaw_live); return

        # ============================================================
        # 状态机
        # ============================================================
        cruise_z = self._takeoff_z - CRUISE_HEIGHT
        yaw_live = self.heading if self.heading is not None else 0.0
        yaw_locked = self._heading0 if self._heading0 is not None else yaw_live

        # 2026-09-11 飞行边界已完全移除 (原 _enforce_boundaries 调用点 → 软墙接管会导致伺服停摆)

        # 全局超时保护
        if not self._land_cmd_sent and sim_t > MISSION_TIMEOUT:
            self._mission_timeout_forced = True
            self.get_logger().error(f"Mission timeout ({MISSION_TIMEOUT}s), forcing land")
            self._enter_step(STEP_LAND)

        # ---- TAKEOFF: 位置保持 → 竖直爬升到巡航高度 ----
        if self.step == STEP_TAKEOFF:
            if self._step_start_time is None:
                self._enter_step(STEP_TAKEOFF)
            elapsed_tk = self._step_elapsed()
            if elapsed_tk < 2.0:
                # 初始位置保持, 让飞机稳定
                self._pub_pos(self._takeoff_x, self._takeoff_y, self._takeoff_z, yaw_live)
            else:
                # 竖直爬升
                climb_elapsed = elapsed_tk - 2.0
                z_ramp = self._takeoff_z - min(CRUISE_HEIGHT, climb_elapsed * TAKEOFF_CLIMB_RATE)
                self._pub_pos(self._takeoff_x, self._takeoff_y, z_ramp, yaw_live)
            if abs(self.z - cruise_z) < 0.5 and elapsed_tk > 4.0:
                self.get_logger().info("Reached cruise height, entering STABILIZE")
                self._enter_step(STEP_STABILIZE)

        # ---- STABILIZE: 悬停稳姿态 → 锁定 heading0 和 home ----
        elif self.step == STEP_STABILIZE:
            # 进入时捕捉固定 yaw, 悬停让 heading 真正收敛
            if self._stabilize_yaw is None:
                self._stabilize_yaw = self.heading if self.heading else 0.0
            self._pub_pos(self._takeoff_x, self._takeoff_y, cruise_z, self._stabilize_yaw)
            if self._step_elapsed() >= STABILIZE_HOVER_TIME:
                # 使用硬编码航向, home 用起飞点地面高度
                self._heading0 = math.radians(HEADING_DEG)
                self.home_x, self.home_y = self._takeoff_x, self._takeoff_y
                self.home_z = self._takeoff_z
                bx, by = self._bucket_target()
                self.get_logger().info(
                    f"Home locked: NED({self.home_x:.2f},{self.home_y:.2f},{self.home_z:.2f}) "
                    f"heading0={HEADING_DEG:.0f}deg (hardcoded) target NED({bx:.1f},{by:.1f})")
                self._enter_step(STEP_FORWARD)

        # ---- FORWARD: 沿 heading0 盲飞 32m, 全程忽略视觉检测, 保证轨迹笔直 ----
        elif self.step == STEP_FORWARD:
            if self._step_elapsed() > STEP_TIMEOUT: self._enter_step(STEP_HOVER_SEARCH); return
            tx, ty = self._bucket_target()
            cos_h0, sin_h0 = math.cos(self._heading0), math.sin(self._heading0)
            # 一维标量投影判断飞行进度 (区分方向: 正=接近, 负=飞过头)
            travel = (self.x - self.home_x) * cos_h0 + (self.y - self.home_y) * sin_h0
            remain = FORWARD_DIST - travel
            # 过冲检测: 飞过目标 0.5m 以上, 连续 N 帧确认 (防 GPS 跳变)
            if remain < -POS_ARRIVE_DIST:
                self._fwd_exit_frames += 1
                if self._fwd_exit_frames >= FORWARD_EXIT_CONFIRM_FRAMES:
                    self.get_logger().info(f"Overshot target travel={travel:.2f}m > {FORWARD_DIST}m, entering HOVER_SEARCH")
                    self._enter_step(STEP_HOVER_SEARCH); return
            else:
                self._fwd_exit_frames = 0
            # 非对称 ramp-down: effective_dist = max(0, remain), 飞过头不加速 (防止越飞越远)
            effective_dist = max(0.0, remain)
            ratio = max(FORWARD_MIN_SPEED / CRUISE_SPEED, min(1.0, effective_dist / FORWARD_RAMP_DIST))
            speed = CRUISE_SPEED * ratio
            vx = speed * cos_h0; vy = speed * sin_h0
            self._pub_vel(vx, vy, (cruise_z - self.z) * TRACK_VZ_KP, self._heading0)
            # 二维欧氏距离兜底 — 连续 N 帧确认
            dist2d = math.hypot((self.x - self.home_x) - tx, (self.y - self.home_y) - ty)
            if dist2d < POS_ARRIVE_DIST:
                self._fwd_arrive_frames += 1
                if self._fwd_arrive_frames >= FORWARD_EXIT_CONFIRM_FRAMES:
                    self.get_logger().info(f"Arrived at target, dist={dist2d:.2f}m, entering HOVER_SEARCH")
                    self._enter_step(STEP_HOVER_SEARCH)
            else:
                self._fwd_arrive_frames = 0

        # ---- HOVER_SEARCH: 目标点悬停, 搜索全部桶, 选最近2个 ----
        elif self.step == STEP_HOVER_SEARCH:
            if self._step_elapsed() > HOVER_SEARCH_TIMEOUT:
                self._search_failed = True
                self.get_logger().info("No valid targets, going to scout zone")
                self._enter_step(STEP_SCOUT); return
            tx, ty = self._bucket_target()
            self._pub_pos(self.home_x + tx, self.home_y + ty, cruise_z, yaw_locked)

            # 驻留等 MPC 刹车, 期间跳过门控
            if self._step_elapsed() < HOVER_SEARCH_SETTLE_TIME:
                return

            # 零检测超时: 连续 N 秒一个桶都没看到 → 判定此处无目标, 转侦察区
            if not self._detect_fresh() or not self._bucket_ests:
                if not hasattr(self, '_zero_detect_start'): self._zero_detect_start = self.get_clock().now()
                if hasattr(self, '_zero_detect_start') and self._zero_detect_start is not None:
                    zero_elapsed = (self.get_clock().now() - self._zero_detect_start).nanoseconds * 1e-9
                    if zero_elapsed > HOVER_SEARCH_ZERO_DETECT_TIMEOUT:
                        self._search_failed = True
                        self.get_logger().info(
                            f"No detection for {zero_elapsed:.1f}s > {HOVER_SEARCH_ZERO_DETECT_TIMEOUT}s, going to scout zone")
                        self._enter_step(STEP_SCOUT); return
            else:
                self._zero_detect_start = None

            # 从已确认的追踪器中选检测框面积最大的 2 个桶
            confirmed = [t for t in self._trackers if t["confirmed"]]
            if len(confirmed) < 2:
                return  # 确认不足 2 个, 继续等待

            # 2026-09-11 选桶策略改版: "检测框面积降序" → "优先选离航向中线最近的桶".
            # 动机 (实机 d_mission_20260911_170203.log): 面积最大 ≠ 最合适 —— 该轮按面积
            # 选中侧向 lateral -4.21m 的桶当第一目标, 几乎正前方的 -0.42m 桶反被丢弃,
            # 迫使飞机做大侧向机动, 并落进广角镜头像场边缘 (定位偏差最大) 的区域.
            # 离 heading0 中线越近: 定位越准、机动越小、飞行风险越低.
            if self._heading0 is not None:
                s_h0, c_h0 = math.sin(self._heading0), math.cos(self._heading0)
                # 主键 |lateral| 升序 (lateral 同 _enforce_boundaries 约定, 正值=heading0 右侧);
                # 次键 面积降序 (两侧向接近时仍优先更近/更大的桶)
                confirmed.sort(key=lambda t: (
                    abs(-t["ned"][0] * s_h0 + t["ned"][1] * c_h0),
                    -t.get("area", 0)))
            else:
                confirmed.sort(key=lambda t: t.get("area", 0), reverse=True)
            selected = [(t["ned"][0], t["ned"][1]) for t in confirmed[:2]]
            lateral_txt = ", ".join(
                f"lat={-t['ned'][0]*math.sin(self._heading0)+t['ned'][1]*math.cos(self._heading0):+.2f}m"
                for t in confirmed[:2]) if self._heading0 is not None else ""
            self._target_buckets = selected
            self._target_count = len(selected)
            self._bucket_idx = 0
            self._current_target_ned = self._target_buckets[self._bucket_idx]
            self._target_img_idx = -1
            self.get_logger().info(
                f"Selected {len(selected)} buckets: "
                + ", ".join(f"NED({bx:.1f},{by:.1f})"
                          for bx, by in selected)
                + (f"  [{lateral_txt}]" if lateral_txt else ""))
            self.pid.reset(); self._enter_step(STEP_TRACK)

        # ---- TRACK: 位置控制飞向已知桶坐标 + 到位后视觉确认 ----
        elif self.step == STEP_TRACK:
            if self._step_elapsed() > STEP_TIMEOUT: self._enter_step(STEP_RETURN); return
            tx, ty = self._current_target_ned
            dist_to_target = math.hypot((self.x - self.home_x) - tx, (self.y - self.home_y) - ty)
            # 全程位置控制, 让 PX4 自己处理导航和刹车
            self._pub_pos(self.home_x + tx, self.home_y + ty, cruise_z, self.heading)
            # 到位判断: 接近目标即可 (视觉确认在到位后)
            if dist_to_target < POS_ARRIVE_DIST:
                self._stabilize_lost_frames = 0
                if self._stabilize_start is None: self._stabilize_start = self.get_clock().now()
                elif (self.get_clock().now()-self._stabilize_start).nanoseconds > STABILIZE_TIME*1e9: self._enter_step(STEP_POSITION_REFINE)
            elif self._step_elapsed() > self._track_max_time: self._enter_step(STEP_POSITION_REFINE)
            else:
                self._stabilize_lost_frames += 1
                if self._stabilize_lost_frames > STABILIZE_LOST_FRAMES_MAX: self._stabilize_start = None; self._stabilize_lost_frames = 0

        # ---- POSITION_REFINE: 悬停稳定后重新估算桶坐标 ----
        elif self.step == STEP_POSITION_REFINE:
            if self._step_elapsed() > 5.0:
                self._finalize_refine()
                self._enter_step(STEP_PRE_SERVO); return

            tx, ty = self._current_target_ned
            self._pub_pos(self.home_x + tx, self.home_y + ty, cruise_z, self.heading)

            if not hasattr(self, '_refine_ests'):
                self._refine_ests = []

            # 几何身份锁: 从当前帧检测中找 NED 估计最接近原始目标的那一个
            if self._bucket_ests:
                best_est = None; best_dist = float('inf')
                for bx, by, roi, d, cx, cy in self._bucket_ests:
                    geo_dist = math.hypot(bx - tx, by - ty)
                    if geo_dist < best_dist:
                        best_dist = geo_dist; best_est = (bx, by)
                if best_est is not None and best_dist < POSITION_REFINE_IDENTITY_MAX:
                    self._refine_ests.append(best_est)
                # 每秒打印候选信息
                if not hasattr(self, '_last_refine_log'):
                    self._last_refine_log = 0.0
                if sim_t - self._last_refine_log > 1.0:
                    self._last_refine_log = sim_t
                    cands = ", ".join(f"({bx:.1f},{by:.1f})d={math.hypot(bx-tx,by-ty):.1f}"
                                     for bx, by, *_ in self._bucket_ests)
                    self.get_logger().info(
                        f"  REFINE candidates vs target({tx:.1f},{ty:.1f}): {cands} "
                        f"best_dist={best_dist:.1f} n={len(self._refine_ests)}")

            if self._step_elapsed() >= POSITION_REFINE_DURATION:
                self._finalize_refine()
                self._enter_step(STEP_PRE_SERVO)

        # ---- PRE_SERVO: 下降到低空提升 pinhole 精度, 再切 VISUAL_SERVO (v7.1) ----
        elif self.step == STEP_PRE_SERVO:
            if self._step_elapsed() > STEP_TIMEOUT: self._enter_step(STEP_VISUAL_SERVO); return
            pre_z = self.home_z - PRE_SERVO_HEIGHT
            # 时间锚定限速下降 (2026-08-15 d): 目标从入口高度以 0.25m/s 匀速下移, min() 封顶.
            # 旧 max(self.z - 0.0125) 增量方向反 (NED z 向下为正) 导致目标瞬间跳变→全速俯冲
            target_z = min(pre_z, self._desc_entry_z + SERVO_DESCENT_SPEED * self._step_elapsed())
            tx, ty = self._current_target_ned
            self._pub_pos(self.home_x + tx, self.home_y + ty, target_z, self.heading)
            if self.z > pre_z - 0.3:
                # 低空了, 用最新 ROI 更新桶坐标 (pinhole 更准), 切 VISUAL_SERVO
                if self._detect_fresh() and self.latest_roi is not None and self._bucket_ests:
                    # 几何身份锁: 找离当前目标最近的检测, 更新 NED 估计
                    best_est = None; best_dist = float('inf')
                    for bx, by, roi, d, cx, cy in self._bucket_ests:
                        geo_dist = math.hypot(bx - tx, by - ty)
                        if geo_dist < best_dist:
                            best_dist = geo_dist; best_est = (bx, by)
                    if best_est is not None and best_dist < 2.0:
                        self._current_target_ned = best_est
                        self.get_logger().info(
                            f"PRE_SERVO update target: ({best_est[0]:.2f},{best_est[1]:.2f}) "
                            f"H={self.home_z-self.z:.1f}m")
                self._enter_step(STEP_VISUAL_SERVO)

        # ---- VISUAL_SERVO: 视觉反投影生成位置设定点；像素误差判断收敛 ----
        elif self.step == STEP_VISUAL_SERVO:
            if self._step_elapsed() > VISUAL_SERVO_TIMEOUT:
                self.get_logger().info("VISUAL_SERVO timeout, dropping")
                self._enter_step(STEP_DROP); return
            drop_z = self.home_z - DROP_HEIGHT
            # 时间锚定限速下降 (2026-08-15 d), 同 PRE_SERVO
            target_z = min(drop_z, self._desc_entry_z + SERVO_DESCENT_SPEED * self._step_elapsed())

            if not self._detect_fresh() or self.latest_roi is None:
                # 丢检: 保持当前伺服目标位置继续下降, 重置抽帧
                self._pub_pos(self.home_x + self._servo_target_x, self.home_y + self._servo_target_y, target_z, self.heading)
                self._decimation_ctr = 0
            else:
                roi = self.latest_roi
                self._decimation_ctr += 1
                if self._decimation_ctr % PID_DECIMATION_SERVO == 0:
                    dt = PID_DECIMATION_SERVO * 0.05
                    dz_h = max(0.3, self.home_z - self.z)
                    offset_body_y = PAYLOAD_OFFSET_BODY_Y_B1 if self._drop_count == 0 else PAYLOAD_OFFSET_BODY_Y_B2
                    pixel_offset = offset_body_y * FX / dz_h
                    height_scale = dz_h / CRUISE_HEIGHT
                    v_fwd, v_lat, err_px, ex, ey = self.pid.compute(
                        roi, dt, pixel_offset, self._pitch, self._roll, height_scale)
                    # 2026-08-31: 锚点+反投影重建 target (废弃增量累加).
                    # 增量伺服在 SITL 低速跟踪失效(方向反偏)时积分漂移正反馈 → 投弹偏 0.6m.
                    # 反投影直接测量桶位置: target 有界无积分器; 锚点做身份锁+平滑.
                    est = self._roi_to_ned(roi)
                    anchor_x, anchor_y = self._current_target_ned
                    if est is not None:
                        est_x, est_y = est
                        if math.hypot(est_x - anchor_x, est_y - anchor_y) < SERVO_GEO_LOCK_MAX:
                            self._servo_target_x = SERVO_EST_WEIGHT * est_x + (1.0 - SERVO_EST_WEIGHT) * anchor_x
                            self._servo_target_y = SERVO_EST_WEIGHT * est_y + (1.0 - SERVO_EST_WEIGHT) * anchor_y
                        else:
                            self._servo_target_x, self._servo_target_y = anchor_x, anchor_y
                    else:
                        self._servo_target_x, self._servo_target_y = anchor_x, anchor_y
                    # 投放补偿: 瞄准点机体横向偏移 (offset>0 → 瞄准点左移, 与 pixel_offset 语义一致)
                    self._servo_target_x += offset_body_y * math.sin(self.heading)
                    self._servo_target_y -= offset_body_y * math.cos(self.heading)
                    # 每秒打印 PID 状态
                    if not hasattr(self, '_last_servo_log'):
                        self._last_servo_log = 0.0
                    if sim_t - self._last_servo_log > 1.0:
                        self._last_servo_log = sim_t
                        self.get_logger().info(
                            f"  SERVO err={err_px:.0f}px ex={ex:.0f} ey={ey:.0f} "
                            f"v_fwd={v_fwd:.3f} v_lat={v_lat:.3f}(log-only) H={dz_h:.1f}m "
                            f"target=({self._servo_target_x:.2f},{self._servo_target_y:.2f}) "
                            f"dec={PID_DECIMATION_SERVO}x")
                    # 收敛检测
                    if err_px < VISUAL_SERVO_ARRIVE_PX:
                        if self._servo_converge_start is None:
                            self._servo_converge_start = self.get_clock().now()
                        elif (self.get_clock().now() - self._servo_converge_start).nanoseconds * 1e-9 > VISUAL_SERVO_CONVERGE_TIME:
                            self.get_logger().info(f"Servo converged! err={err_px:.0f}px, dropping")
                            self._enter_step(STEP_DROP)
                    else:
                        self._servo_converge_start = None
                # 非抽帧: 重发当前位置维持 offboard >2Hz
                self._pub_pos(self.home_x + self._servo_target_x, self.home_y + self._servo_target_y, target_z, self.heading)

            # 接近投放高度进入 DROP，继续视觉保持与释放条件判断
            if self.z > drop_z - 0.2:
                self._enter_step(STEP_DROP)

        # ---- DROP: 投弹 ----
        elif self.step == STEP_DROP:
            if self._drop_triggered:
                # 释放后固定本次设定点，独立计时；不读取下一瓶补偿或再次判断收敛。
                self._pub_pos(*self._drop_hold_target)
                held_s = (self.get_clock().now() - self._drop_signal_start).nanoseconds * 1e-9
                if held_s >= AUX5_DROP_HOLD:
                    self._aux5_value = AUX5_LOCK
                    self.get_logger().info(f"DROP signal reset after {held_s:.2f}s (command hold, no mechanism feedback)")
                    self._enter_step(STEP_CLIMB)
                return
            if self._step_elapsed() > STEP_TIMEOUT: self._aux5_value = AUX5_LOCK; self._enter_step(STEP_CLIMB); return
            drop_z = self.home_z - DROP_HEIGHT
            if self._servo_active and self._drop_detection_fresh():
                roi = self.latest_roi
                if roi is not None:
                    self._decimation_ctr += 1
                    if self._decimation_ctr % PID_DECIMATION_DROP == 0:
                        dt = PID_DECIMATION_DROP * 0.05
                        dz_h = max(0.3, self.home_z - self.z)
                        offset_body_y = PAYLOAD_OFFSET_BODY_Y_B1 if self._drop_count == 0 else PAYLOAD_OFFSET_BODY_Y_B2
                        pixel_offset = offset_body_y * FX / dz_h
                        height_scale = dz_h / CRUISE_HEIGHT
                        v_fwd, v_lat, err_px, ex, ey = self.pid.compute(
                            roi, dt, pixel_offset, self._pitch, self._roll, height_scale)
                        self._last_drop_err = (err_px, ex, ey, dz_h)
                        # 2026-08-31: 锚点+反投影重建 target (同 VISUAL_SERVO, 防漂移)
                        est = self._roi_to_ned(roi)
                        anchor_x, anchor_y = self._current_target_ned
                        lock_mode = 'no_est'; lock_dist = float('nan')
                        est_x = est_y = float('nan')
                        if est is not None:
                            est_x, est_y = est
                            lock_dist = math.hypot(est_x - anchor_x, est_y - anchor_y)
                            # 2026-09-13 身份锁改造 (实机 9-13 投弹偏差修复): 原"超门限即冻结在锚点"会把随高度变化的
                            # 反投影漂移 (同一静止桶 5m→1.5m 漂 0.65m) 判成误检 → target 永远停在旧锚点 → err 不收敛
                            # → 只能 25s 兜底投弹, 落点偏 0.235m. 改为"连续一致即重锚定"(疑桶仍由 reanchor_max 挡住).
                            lock_mode, lx, ly, self._geo_reject_n, self._geo_pending_est = geo_lock_decide(
                                est_x, est_y, anchor_x, anchor_y, SERVO_GEO_LOCK_MAX, SERVO_EST_WEIGHT,
                                self._geo_pending_est, self._geo_reject_n,
                                SERVO_GEO_REANCHOR_N, SERVO_GEO_REANCHOR_RAD, SERVO_GEO_REANCHOR_MAX)
                            if lock_mode == 'reanchor':
                                # 认可新位置: 锚点同步更新 (此后检测不新鲜时回退目标也用新锚点)
                                self._current_target_ned = (est_x, est_y)
                                anchor_x, anchor_y = est_x, est_y
                            self._servo_target_x, self._servo_target_y = lx, ly
                        else:
                            self._servo_target_x, self._servo_target_y = anchor_x, anchor_y
                        if lock_mode != self._geo_mode:
                            self._geo_mode = lock_mode
                            self.get_logger().info(
                                f"  DROP-LOCK {lock_mode} est=({est_x:.2f},{est_y:.2f}) "
                                f"anchor=({anchor_x:.2f},{anchor_y:.2f}) dist={lock_dist:.2f}m "
                                f"reject_n={self._geo_reject_n} err={err_px:.0f}px")
                        # 2026-09-13 像素残差积分 (闭环项): v_fwd/v_lat 是像素 PID 输出, 原本只打日志 → DROP 段
                        # 是"纯反投影开环", 反投影的系统性偏置 (内参/镜头/框中心定义) 100% 变成投弹偏差. 这里把它
                        # 的积分加到重建目标上, 让 err_px 真正收敛 → 释放瞬间桶在画面中心 (不依赖地面标定).
                        # 只在身份锁采信/重锚 (确认还是同一个桶) 且像素偏差合理时积分, 防误检把目标拖偏.
                        if (lock_mode in ('accept', 'reanchor')
                                and math.hypot(ex * dz_h / FX, ey * dz_h / FY) < SERVO_BIAS_ERR_MAX_M
                            and math.hypot(self.vx, self.vy) < SERVO_BIAS_VMAX):
                            # 低空 height_scale(=H/CRUISE_HEIGHT=0.3) 会把速率压到 1/3.3, 部分反除 (最多 2x):
                            # 该缩放是为防 bbox jitter 引起的高频振荡, 而积分项本身低通 (jitter 被平均掉);
                            # 但不反除则 0.24m 偏置要 ~17s 才收敛. 实测 3.3x 反除会与内环迟滞绕圈振荡
                            # (2026-09-13 SITL d 首投 target ±0.4m 摆振 → 0.236m/6分), 故 2x + 准静止门控.
                            sc = 1.0 / max(0.5, height_scale)
                            bvf = max(-MAX_HORIZ_VEL, min(MAX_HORIZ_VEL, v_fwd * sc))
                            bvl = max(-MAX_HORIZ_VEL, min(MAX_HORIZ_VEL, v_lat * sc))
                            bdx, bdy, _ = self._body_to_ned(bvf * dt, bvl * dt)
                            self._servo_bias_x += bdx
                            self._servo_bias_y += bdy
                            bias_norm = math.hypot(self._servo_bias_x, self._servo_bias_y)
                            if bias_norm > SERVO_BIAS_MAX:
                                k = SERVO_BIAS_MAX / bias_norm
                                self._servo_bias_x *= k
                                self._servo_bias_y *= k
                                if not self._bias_sat_logged:
                                    self._bias_sat_logged = True
                                    self.get_logger().warn(
                                        f"DROP bias saturated {SERVO_BIAS_MAX:.2f}m (err={err_px:.0f}px) "
                                        f"— 反投影偏置超预期, 建议实机地面标定相机内参")
                        self._servo_target_x += self._servo_bias_x
                        self._servo_target_y += self._servo_bias_y
                        # 两路载荷横向偏移，保留用户现有实机配置。
                        self._servo_target_x += offset_body_y * math.sin(self.heading)
                        self._servo_target_y -= offset_body_y * math.cos(self.heading)
                        # DROP 保持期伺服明细 (2026-08-15 d), 观察延时内收敛情况
                        if not hasattr(self, '_last_servo_log'): self._last_servo_log = 0.0
                        if sim_t - self._last_servo_log > 1.0:
                            self._last_servo_log = sim_t
                            self.get_logger().info(
                                f"  DROP-SERVO err={err_px:.0f}px ex={ex:.0f} ey={ey:.0f} "
                                f"v_fwd={v_fwd:.3f} v_lat={v_lat:.3f}(bias-input) H={dz_h:.1f}m "
                                f"target=({self._servo_target_x:.2f},{self._servo_target_y:.2f}) "
                                f"dec={PID_DECIMATION_DROP}x "
                            f"lock={lock_mode} dist={lock_dist:.2f}m "
                            f"bias=({self._servo_bias_x:+.2f},{self._servo_bias_y:+.2f}) pitch={self._pitch:+.3f} roll={self._roll:+.3f} "
                            f"vxy={math.hypot(self.vx, self.vy):.3f}m/s det_age={self._drop_detect_age():.3f}s "
                            f"等效地面={math.hypot(ex * dz_h / FX, ey * dz_h / FY) * 100:.1f}cm(门限{DROP_ARRIVE_M * 100:.0f}cm)")
                self._pub_pos(self.home_x + self._servo_target_x, self.home_y + self._servo_target_y, drop_z, self.heading)
            else:
                self._last_drop_err = None
                self._drop_converge_start = None
                self._decimation_ctr = 0
                self._pub_pos(self.home_x + self._servo_target_x, self.home_y + self._servo_target_y, drop_z, self.heading)
            # 2026-08-31: 收敛即投 — err 折算"等效地面距离"<DROP_ARRIVE_M(10cm) 持续 DROP_CONVERGE_EARLY 则提前投弹;
            # 2026-08-31 审查修复: 加高度门控 — VISUAL_SERVO 收敛可在任意高度切 DROP (实测 H=2.23m),
            # 若只判 err 会高空投弹; 要求已降到投弹高度 (同"高度到达切 DROP"判据 H<drop_z+0.2 = 1.4m @DROP_HEIGHT 1.2)
            # 否则 DROP_AFTER_DELAY 兜底 (实机未收敛场景)
            # 2026-09-18: 收敛判据改米制 —— 把最近一帧 (ex, ey, H) 折算成"等效地面距离"
            conv_m = float('inf')
            if self._last_drop_err is not None:
                _ep, _exl, _eyl, _hh = self._last_drop_err
                conv_m = math.hypot(_exl * _hh / FX, _eyl * _hh / FY)
            early_ok = False
            release_valid = (self._drop_detection_fresh() and math.isfinite(conv_m)
                             and self.z > drop_z - 0.2)
            if release_valid and conv_m < DROP_ARRIVE_M:
                if self._drop_converge_start is None:
                    self._drop_converge_start = sim_t
                elif sim_t - self._drop_converge_start > DROP_CONVERGE_EARLY:
                    early_ok = True
            else:
                self._drop_converge_start = None
            if not early_ok and self._step_elapsed() < DROP_AFTER_DELAY: return
            if not release_valid:
                self.get_logger().warn("DROP timeout without fresh valid geometry/height; skip release and climb")
                self._aux5_value = AUX5_LOCK
                self._enter_step(STEP_CLIMB)
                return
            if not self._drop_triggered:
                self._drop_triggered = True; self._drop_signal_start = self.get_clock().now()
                self._drop_hold_target = (
                    self.home_x + self._servo_target_x, self.home_y + self._servo_target_y,
                    drop_z, self.heading)
                # B1→2000us, B2→1000us
                drop_val = AUX5_DROP if self._drop_count == 0 else AUX5_DROP2
                pwm_us = 1500 + drop_val * 600
                self._aux5_value = drop_val
                self._drop_count += 1
                # 兼容已有话题名：这里只发布指令时飞机位置，不是测得的落点。
                self.impact_pub.publish(Point(x=float(self.x), y=float(self.y), z=float(self.home_z)))
                err_str = ""
                if self._last_drop_err is not None:
                    ep, exl, eyl, hh = self._last_drop_err
                    err_str = (f" last_err={ep:.0f}px(ex={exl:.0f},ey={eyl:.0f}@H={hh:.1f}m)"
                               f" 等效地面={math.hypot(exl * hh / FX, eyl * hh / FY) * 100:.1f}cm")
                drop_way = ("收敛即投" if early_ok
                            else f"{self._step_elapsed():.1f}s/{DROP_AFTER_DELAY:.0f}s 兜底")
                self._drop_records.append({
                    "idx": self._drop_count,
                    # 2026-08-31 审查修复: aim 存绝对 NED (与 pos 同系).
                    # 原 tuple(_current_target_ned) 是相对 home 偏移, 与绝对 pos 相减算 dev 会虚高 |home|
                    "aim": (self.home_x + self._current_target_ned[0],
                            self.home_y + self._current_target_ned[1]) if self._current_target_ned else None,
                    "pos": (float(self.x), float(self.y), float(self.z)),
                    "H": float(self.home_z - self.z),
                    "last_err": self._last_drop_err,
                })
                self.get_logger().info(
                    f">>> DROP #{self._drop_count}! PWM={pwm_us:.0f}us NED({self.x:.3f},{self.y:.3f},{self.z:.3f}) "
                    f"heading={math.degrees(self.heading):.1f}deg 触发方式={drop_way}{err_str} "
                    f"pitch={self._pitch:+.3f} roll={self._roll:+.3f} "
                    f"vxy={math.hypot(self.vx, self.vy):.3f}m/s det_age={self._drop_detect_age():.3f}s "
                    f"hold_target={self._drop_hold_target} event=release_command"); return

        # ---- CLIMB: 投弹后爬升 → 下一桶或侦察区 ----
        elif self.step == STEP_CLIMB:
            if self._step_elapsed() > STEP_TIMEOUT: self._enter_step(STEP_RETURN); return
            # 阶段1: 竖直爬升到巡航高度
            if not hasattr(self, '_climb_done'): self._climb_done = False
            if not self._climb_done:
                vz = CLIMB_VZ + (cruise_z-self.z)*TRACK_VZ_KP
                self._pub_vel(0, 0, vz, yaw_locked)
                if abs(self.z-cruise_z) < 0.5:
                    self._climb_done = True
                    self._bucket_idx += 1
                return
            # 阶段2: 直接进 TRACK 飞向下一个桶 (用最初锁定的坐标, 不重搜)
            self._climb_done = False
            if self._bucket_idx < len(self._target_buckets):
                self._current_target_ned = self._target_buckets[self._bucket_idx]
                self._target_img_idx = -1

                    f"Next bucket {self._bucket_idx+1}/{len(self._target_buckets)}: "
                    f"NED({self._current_target_ned[0]:.1f},{self._current_target_ned[1]:.1f})")

            else:
                self.get_logger().info("All buckets done, flying to SCOUT zone")
                self._enter_step(STEP_SCOUT)

        # ---- SCOUT (2026-08-15 重写, 合并自 c): 直飞近边 → 降1m → 10s缓移至预定点 → 左1.5m → 停5s → 右3m → 停5s → 返航 ----
        elif self.step == STEP_SCOUT:
            if self._step_elapsed() > STEP_TIMEOUT: self._enter_step(STEP_RETURN); return
            sx, sy = self._scout_target()       # 预定点 (侦察区正上方)
            ex, ey = self._scout_edge_target()   # 第一根线(近边)中点
            if self._scout_sub == 0:
                # 子阶段0: 速度控制直飞近边 (不再直冲预定点, 防过冲出侦察区)
                dist_to_edge = math.hypot((self.x - self.home_x) - ex, (self.y - self.home_y) - ey)
                if dist_to_edge > POS_ARRIVE_DIST:
                    vx, vy, _ = self._fly_toward(self.home_x + ex, self.home_y + ey, CRUISE_SPEED)
                    self._pub_vel(vx, vy, (cruise_z - self.z) * TRACK_VZ_KP, yaw_locked)
                    return
                self._scout_sub = 1
                self._scout_sub_start = self.get_clock().now()
                self.get_logger().info("Scout: arrived at zone near edge, descending 1m first")
                return
            sub_elapsed = (self.get_clock().now() - self._scout_sub_start).nanoseconds * 1e-9
            # 机体左/右向单位向量 (基于锁定航向)
            left_x, left_y = math.sin(yaw_locked), -math.cos(yaw_locked)
            scout_z = cruise_z + SCOUT_DESCENT_DELTA
            if self._scout_sub == 1:
                # 子阶段1: 先下降 1m (5m→4m)
                self._pub_pos(self.home_x + ex, self.home_y + ey, scout_z, yaw_locked)
                if sub_elapsed > SCOUT_SUB_TIMEOUT:
                    self.get_logger().warn("Scout descend timeout, starting slow move")
                    self._scout_sub = 2; self._scout_sub_start = self.get_clock().now(); return
                if abs(self.z - scout_z) < SCOUT_SUB_ARRIVE:
                    self._scout_sub = 2; self._scout_sub_start = self.get_clock().now()
                    self.get_logger().info(
                        f"Scout: descended 1m, slow moving {SCOUT_SLOW_MOVE_TIME:.0f}s to zone center")
                return
            if self._scout_sub == 2:
                # 子阶段2: 10s 缓慢移动到预定点 (时间锚定线性插值, 0.25m/s 防过冲; SITL 版 0.45m/s)
                t = min(1.0, sub_elapsed / SCOUT_SLOW_MOVE_TIME)
                tx = ex + (sx - ex) * t
                ty = ey + (sy - ey) * t
                self._pub_pos(self.home_x + tx, self.home_y + ty, scout_z, yaw_locked)
                if sub_elapsed >= SCOUT_SLOW_MOVE_TIME:
                    self._scout_sub = 3; self._scout_sub_start = self.get_clock().now()
                    self.get_logger().info("Scout: slow move done, moving left 1.5m")
                return
            if self._scout_sub == 3:
                # 子阶段3: 机体左移 1.5m
                tx, ty = sx + left_x * SCOUT_LEFT_DIST, sy + left_y * SCOUT_LEFT_DIST
                self._pub_pos(self.home_x + tx, self.home_y + ty, scout_z, yaw_locked)
                if sub_elapsed > SCOUT_SUB_TIMEOUT:
                    self.get_logger().warn("Scout left-move timeout, pausing")
                    self._scout_sub = 4; self._scout_sub_start = self.get_clock().now(); return
                if math.hypot((self.x - self.home_x) - tx, (self.y - self.home_y) - ty) < SCOUT_SUB_ARRIVE:
                    self._scout_sub = 4; self._scout_sub_start = self.get_clock().now()
                    self.get_logger().info("Scout: left 1.5m done, pausing 5s")
                return

                if sub_elapsed >= SCOUT_PAUSE_TIME:
                    self._scout_sub = 5; self._scout_sub_start = self.get_clock().now()
                    self.get_logger().info("Scout: pause done, moving right 3m")
                return

                tx = sx + left_x * SCOUT_LEFT_DIST - left_x * SCOUT_RIGHT_DIST
                ty = sy + left_y * SCOUT_LEFT_DIST - left_y * SCOUT_RIGHT_DIST

                if sub_elapsed > SCOUT_SUB_TIMEOUT:
                    self.get_logger().warn("Scout right-move timeout, pausing")
                    self._scout_sub = 6; self._scout_sub_start = self.get_clock().now(); return
                if math.hypot((self.x - self.home_x) - tx, (self.y - self.home_y) - ty) < SCOUT_SUB_ARRIVE:

                return

            ty = sy + left_y * SCOUT_LEFT_DIST - left_y * SCOUT_RIGHT_DIST
            self._pub_pos(self.home_x + tx, self.home_y + ty, scout_z, yaw_locked)
            if sub_elapsed >= SCOUT_PAUSE_TIME:
                self._scout_done = True
                self._scout_elapsed = self._step_elapsed()
                self.get_logger().info("Scout maneuver complete, returning home")
                self._enter_step(STEP_RETURN)

        # ---- RETURN (2026-08-15, 合并自 c): 飞回 home → 位置保持判稳 (漂移<0.3m/3s) → LAND ----
        elif self.step == STEP_RETURN:
            if self._step_elapsed() > STEP_TIMEOUT: self._enter_step(STEP_LAND); return

                self._pub_vel(vx, vy, (cruise_z-self.z)*TRACK_VZ_KP, yaw_locked)

                    self._return_sub = 1
                    self._return_sub_start = self.get_clock().now()
                    self.get_logger().info("Return: over home, stabilizing before land")
                return

            sub_elapsed = (self.get_clock().now() - self._return_sub_start).nanoseconds * 1e-9
            self._pub_pos(self.home_x, self.home_y, cruise_z, yaw_locked)
            self._stab_pos_hist.append((self.x, self.y))

                self._enter_step(STEP_LAND); return

                ys = [p[1] for p in self._stab_pos_hist]

                    self._return_stab_info = (sub_elapsed, drift)

                        f"Return: stable (drift {drift:.2f}m over {RETURN_STAB_WINDOW:.0f}s), landing")
                    self._enter_step(STEP_LAND)

        # ---- LAND (2026-08-15, 合并自 c: 时间锚定限速下降 0.4m/s, 防降太猛) ----
        elif self.step == STEP_LAND:
            if self._step_elapsed() > STEP_TIMEOUT:
                self.get_logger().error("LAND step timeout (120s), forcing disarm")
                self._land_pos = (self.x, self.y)
                self._send_cmd(400, param1=0.0); self._shutdown("land_step_timeout"); return

                if self._land_entry_z is None:
                    self._land_entry_z = self.z
                # 时间锚定: 目标从入口高度以 LAND_DESCENT_SPEED 匀速下移, min() 封顶 (独立于飞机实时位置)
                target_z = min(low_z, self._land_entry_z + LAND_DESCENT_SPEED * self._step_elapsed())
                self._pub_pos(self.home_x, self.home_y, target_z, yaw_locked)
                if abs(self.z-low_z) < 0.5:
                    self._land_cmd_sent = True
                    self._land_cmd_time = self.get_clock().now()
                    self.get_logger().info(
                        f"Sending MAV_CMD_NAV_LAND at NED({self.x:.2f},{self.y:.2f},{self.z:.2f}) "
                        f"H={self.home_z-self.z:.2f}m (descent {self._step_elapsed():.1f}s)")
                    self._send_cmd(21, param1=0.0, param2=0.0, param7=float('nan'))
            elif (self.get_clock().now() - self._land_cmd_time).nanoseconds * 1e-9 > 15.0:
                self.get_logger().info("LAND timeout after cmd, forcing disarm")
                self._land_pos = (self.x, self.y)
                self._send_cmd(400, param1=0.0); self._shutdown("land_timeout"); return
            else:
                if self.landed:

                        self._send_cmd(400, param1=0.0); self._shutdown("normal")

"""Run the unmodified DMission state machine from takeoff to landing over ROS 2.

The vehicle, camera, land detector and command acknowledgments are synthetic.
This is a full mission software simulation, not PX4/Gazebo SITL.
"""
import argparse
import json
import math
from pathlib import Path
import time

import rclpy
from rclpy.executors import SingleThreadedExecutor
from px4_msgs.msg import VehicleCommand, VehicleLandDetected, VehicleControlMode
from std_msgs.msg import String

from sim_ros2_buckets import SyntheticPX4, QOS_BE, DT, d, render_roi


class FullMissionVehicle(SyntheticPX4):
    def __init__(self):
        super().__init__()
        self.z = 0.0
        self.targets = [(-34.2, 0.6), (-34.2, -1.8)]
        self.armed = False
        self.offboard = False
        self.land_commanded = False
        self.disarmed = False
        self.commands = []
        self.land_pub = self.create_publisher(VehicleLandDetected, d.LAND_DETECTED_TOPIC, QOS_BE)
        self.command_sub = self.create_subscription(
            VehicleCommand, "/fmu/in/vehicle_command", self.on_command, 10)

    def on_command(self, msg):
        command = int(msg.command)
        now = round(time.monotonic() - self.start_time, 2)
        if command == 400:
            self.armed = msg.param1 > 0.5
            if not self.armed:
                self.disarmed = True
                self.commands.append({"command": "DISARM", "time_s": now})
            else:
                self.commands.append({"command": "ARM", "time_s": now})
        elif command == 176:
            self.offboard = True
            self.commands.append({"command": "OFFBOARD", "time_s": now})
        elif command == 21:
            self.land_commanded = True
            self.commands.append({"command": "LAND", "time_s": now})

    def _integrate_vehicle(self):
        if self.land_commanded:
            self.vx *= .9
            self.vy *= .9
            self.vz = min(.55, max(0., -self.z / DT))
            self.x += self.vx * DT
            self.y += self.vy * DT
            self.z = min(0., self.z + self.vz * DT)
        else:
            super()._integrate_vehicle()

    def _publish_sensors(self):
        super()._publish_sensors()
        stamp = self.get_clock().now().nanoseconds // 1000
        mode = VehicleControlMode()
        mode.timestamp = stamp
        mode.flag_armed = self.armed
        mode.flag_control_offboard_enabled = self.offboard
        self.control_pub.publish(mode)
        land = VehicleLandDetected()
        land.timestamp = stamp
        land.landed = bool(self.z >= -.06)
        self.land_pub.publish(land)

    def _publish_detection(self, now):
        if self._tick_count % 2 or now < self.blackout_until:
            return
        buckets = []
        height = -self.z
        if height > .3:
            for tx, ty in self.targets:
                dx, dy = tx-self.x, ty-self.y
                fwd = dx*math.cos(self.heading)+dy*math.sin(self.heading)
                right = -dx*math.sin(self.heading)+dy*math.cos(self.heading)
                roi = render_roi(fwd, right, height, self.pitch, self.roll)
                cx = roi.x_offset+10
                cy = roi.y_offset+10
                if 10 <= cx <= 630 and 10 <= cy <= 350:
                    buckets.append({"x": round(roi.x_offset), "y": round(roi.y_offset),
                                    "w": 20, "h": 20})
        self.detect_pub.publish(String(data=json.dumps({"buckets": buckets})))


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--timeout", type=float, default=300.)
    args = parser.parse_args()
    rclpy.init()
    mission = d.DMission()
    sim = FullMissionVehicle()
    sim.mission = mission
    executor = SingleThreadedExecutor()
    executor.add_node(sim)
    executor.add_node(mission)
    try:
        while time.monotonic() - sim.start_time < args.timeout:
            executor.spin_once(timeout_sec=.02)
            if sim.disarmed:
                # Allow the mission to finish its normal shutdown and print its summary.
                executor.spin_once(timeout_sec=.1)
                break
        required = ["TAKEOFF", "STABILIZE", "FORWARD", "HOVER_SEARCH",
                    "TRACK", "POSITION_REFINE", "PRE_SERVO", "VISUAL_SERVO",
                    "DROP", "CLIMB", "SCOUT", "RETURN", "LAND"]
        order = all(name in sim.step_history for name in required)
        drops = [e for e in sim.events if e.get("event") == "release_command"]
        summary = {
            "kind": "Full mission ROS2 closed-loop software simulation; synthetic vehicle/camera, no PX4 SITL",
            "code": str((Path(__file__).resolve().parent.parent / "d.py")),
            "elapsed_s": round(time.monotonic()-sim.start_time, 2),
            "step_history": sim.step_history,
            "required_step_order_observed": order,
            "events": sim.events,
            "commands": sim.commands,
            "drop_commands": mission._drop_count,
            "release_events": drops,
            "aux_holds_s": [round(x, 3) for x in sim.aux_holds],
            "first_hold_target_changed": sim.held_setpoint_changed,
            "scout_done": mission._scout_done,
            "return_stabilized": mission._return_stab_info is not None,
            "land_commanded": sim.land_commanded,
            "disarmed": sim.disarmed,
            "land_position_ned": mission._land_pos,
            "home_position_ned": [mission.home_x, mission.home_y],
        }
        summary["passed"] = (order and mission._drop_count == 2
            and len(sim.aux_holds) == 2 and all(.9 <= x <= 1.3 for x in sim.aux_holds)
            and not sim.held_setpoint_changed and mission._scout_done
            and mission._return_stab_info is not None and sim.land_commanded
            and sim.disarmed and mission._land_pos is not None
            and len(drops) == 2
            and all(e["horizontal_target_error_m"] <= .12 for e in drops))
        args.output.parent.mkdir(parents=True, exist_ok=True)
        args.output.write_text(json.dumps(summary, ensure_ascii=False, indent=2), encoding="utf-8")
        print(json.dumps(summary, ensure_ascii=False, indent=2), flush=True)
        if not summary["passed"]:
            raise SystemExit(1)
    finally:
        try:
            executor.remove_node(mission)
        except Exception:
            pass
        executor.remove_node(sim)
        sim.destroy_node()
        try:
            mission.destroy_node()
        except Exception:
            pass
        rclpy.shutdown()


if __name__ == "__main__":
    main()

def quaternion(roll, pitch, yaw):
    cr, sr = math.cos(roll/2), math.sin(roll/2)

    return [cr*cp*cy + sr*sp*sy,
            sr*cp*cy - cr*sp*sy,
            cr*sp*cy + sr*cp*sy,
            cr*cp*sy - sr*sp*cy]

    def on_setpoint(self, msg):
        self.setpoint = msg
        if self.aux_active and len(self.aux_holds) == 0 and self.first_hold_target is not None:
            if all(math.isfinite(v) for v in msg.position[:2]):
                err = math.hypot(msg.position[0]-self.first_hold_target[0],
                                 msg.position[1]-self.first_hold_target[1])
                if err > .015:
                    self.held_setpoint_changed = True

    def on_manual(self, msg):
        now = time.monotonic() - self.start_time
        high = abs(msg.aux5) > .5
        if high and not self.aux_active:
            self.aux_active = True
            self.aux_started = now
            if len(self.aux_holds) == 0:
                self.blackout_until = now + 1.45  # 第一瓶指令后故意切断检测，覆盖 1s 保持
                self.first_hold_target = None
        elif not high and self.aux_active:
            self.aux_active = False
            self.aux_holds.append(now-self.aux_started)
            self.events.append({"event": "aux_reset", "time_s": round(now, 2),
                                "hold_s": round(now-self.aux_started, 3)})

    def _integrate_vehicle(self):
        if self.setpoint is None:
            return
        p, v = self.setpoint.position, self.setpoint.velocity
        if all(math.isfinite(x) for x in p):
            cmd = (clamp(1.4*(p[0]-self.x), .65),
                   clamp(1.4*(p[1]-self.y), .65),
                   clamp(1.4*(p[2]-self.z), .42))
        elif all(math.isfinite(x) for x in v):
            cmd = (clamp(v[0], 2.), clamp(v[1], 2.), clamp(v[2], 1.8))
        else:
            return
        a = DT/(.35+DT)
        self.vx += a*(cmd[0]-self.vx)
        self.vy += a*(cmd[1]-self.vy)
        self.vz += a*(cmd[2]-self.vz)
        self.x += self.vx*DT
        self.y += self.vy*DT
        self.z += self.vz*DT

    def _publish_sensors(self):
        stamp = self.get_clock().now().nanoseconds // 1000
        local = VehicleLocalPosition()
        local.timestamp = stamp
        local.timestamp_sample = stamp
        local.x, local.y, local.z = self.x, self.y, self.z
        local.vx, local.vy, local.vz = self.vx, self.vy, self.vz
        local.heading = self.heading
        local.xy_valid = local.z_valid = True
        self.local_pub.publish(local)
        mode = VehicleControlMode()
        mode.timestamp = stamp
        mode.flag_armed = True
        mode.flag_control_offboard_enabled = True
        self.control_pub.publish(mode)
        attitude = VehicleAttitude()
        attitude.timestamp = attitude.timestamp_sample = stamp
        attitude.q = quaternion(self.roll, self.pitch, self.heading)
        self.attitude_pub.publish(attitude)

    def _publish_detection(self, now):
        if self._tick_count % 2 or now < self.blackout_until:
            return
        idx = min(self.mission._bucket_idx, len(self.targets)-1)
        tx, ty = self.targets[idx]
        dx, dy = tx-self.x, ty-self.y
        fwd = dx*math.cos(self.heading)+dy*math.sin(self.heading)
        right = -dx*math.sin(self.heading)+dy*math.cos(self.heading)
        buckets = []
        height = -self.z
        if height > .3:
            roi = render_roi(fwd, right, height, self.pitch, self.roll)
            cx = roi.x_offset+10
            cy = roi.y_offset+10
            if 10 <= cx <= 630 and 10 <= cy <= 350:
                buckets.append({"x": round(roi.x_offset), "y": round(roi.y_offset),
                                "w": 20, "h": 20})
        self.detect_pub.publish(String(data=json.dumps({"buckets": buckets})))

    def tick(self):
        self._tick_count += 1
        now = time.monotonic()-self.start_time
        self._integrate_vehicle()
        idx = min(self.mission._bucket_idx, 1)
        self.pitch = (.055 if idx == 0 else .02) + .008*math.sin(now*.55)
        self.roll = .02*math.sin(now*.7)
        self._publish_sensors()
        self._publish_detection(now)
        if self.mission.step != self.last_step:
            name = d.STEP_NAME[self.mission.step]
            self.events.append({"event": "step", "name": name, "time_s": round(now, 2)})
            self.step_history.append(name)
            self.last_step = self.mission.step
        if self.mission._drop_count != self.last_drop_count:
            self.last_drop_count = self.mission._drop_count
            offset = d.PAYLOAD_OFFSET_BODY_Y_B1 if self.last_drop_count == 1 else d.PAYLOAD_OFFSET_BODY_Y_B2
            tx, ty = self.targets[self.last_drop_count-1]
            dx, dy = tx-self.x, ty-self.y
            fwd = dx*math.cos(self.heading)+dy*math.sin(self.heading)
            right = -dx*math.sin(self.heading)+dy*math.cos(self.heading)
            self.events.append({"event": "release_command", "index": self.last_drop_count,
                                "time_s": round(now, 2), "height_m": round(-self.z, 3),
                                "horizontal_target_error_m": round(math.hypot(fwd, right-offset), 3),
                                "pitch_rad": round(self.pitch, 4), "roll_rad": round(self.roll, 4)})
            if self.last_drop_count == 1:
                self.first_hold_target = self.mission._drop_hold_target


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--output", type=Path, required=True)
    parser.add_argument("--timeout", type=float, default=70.)
    args = parser.parse_args()
    rclpy.init()
    mission = d.DMission()
    sim = SyntheticPX4()
    sim.mission = mission
    mission.x, mission.y, mission.z = sim.x, sim.y, sim.z
    mission.heading = sim.heading
    mission._takeoff_x = mission._takeoff_y = mission._takeoff_z = 0.
    mission.home_x = mission.home_y = mission.home_z = 0.
    mission._heading0 = sim.heading
    mission._start_wall = mission.get_clock().now()
    mission._initial_wait_done = True
    mission._mode_req_start = mission.get_clock().now()
    mission._arm_offboard_logged = True
    mission._target_buckets = sim.targets[:]
    mission._target_count = 2
    mission._bucket_idx = 0
    mission._current_target_ned = sim.targets[0]
    mission._enter_step(d.STEP_VISUAL_SERVO)
    executor = SingleThreadedExecutor()
    executor.add_node(sim)
    executor.add_node(mission)
    try:
        while time.monotonic()-sim.start_time < args.timeout:
            executor.spin_once(timeout_sec=.02)
            if mission._drop_count >= 2 and len(sim.aux_holds) >= 2:
                break
        summary = {
            "kind": "ROS2 closed-loop software simulation; synthetic vehicle and camera, no PX4 SITL",
            "code": str((HERE.parent / "d.py").resolve()),
            "elapsed_s": round(time.monotonic()-sim.start_time, 2),
            "events": sim.events,
            "drop_commands": mission._drop_count,
            "aux_holds_s": [round(x, 3) for x in sim.aux_holds],
            "first_hold_target_changed": sim.held_setpoint_changed,
            "passed": (mission._drop_count == 2 and len(sim.aux_holds) == 2
                       and all(.9 <= x <= 1.3 for x in sim.aux_holds)
                       and not sim.held_setpoint_changed),
        }
        args.output.parent.mkdir(parents=True, exist_ok=True)
        args.output.write_text(json.dumps(summary, ensure_ascii=False, indent=2), encoding="utf-8")
        print(json.dumps(summary, ensure_ascii=False, indent=2), flush=True)
        if not summary["passed"]:
            raise SystemExit(1)
    finally:
        executor.remove_node(mission)
        executor.remove_node(sim)
        sim.destroy_node()
        mission.destroy_node()
        rclpy.shutdown()


if __name__ == "__main__":
    main()

def quat_rotate(q, v):
    """独立的四元数参考计算，与被测代码的欧拉逐轴实现分开。"""
    w, x, y, z = q
    a, b, c = v
    t = (2 * (y*c-z*b), 2 * (z*a-x*c), 2 * (x*b-y*a))
    return (a+w*t[0]+y*t[2]-z*t[1],
            b+w*t[1]+z*t[0]-x*t[2],
            c+w*t[2]+x*t[1]-y*t[0])


def render_roi(fwd, right, height, pitch=0., roll=0.):
    cp, sp = math.cos(pitch/2), math.sin(pitch/2)
    cr, sr = math.cos(roll/2), math.sin(roll/2)
    q_body_to_level = (cr*cp, sr*cp, cr*sp, -sr*sp)
    w, x, y, z = q_body_to_level
    bx, by, bz = quat_rotate((w, -x, -y, -z), (fwd, right, height))
    cx = env["CX"] + env["FX"] * by/bz
    cy = env["CY"] - env["FY"] * bx/bz
    return SimpleNamespace(x_offset=cx-10, y_offset=cy-10, width=20, height=20)


class Time:
    def __init__(self, seconds):
        self.nanoseconds = round(seconds * 1e9)

    def __sub__(self, other):
        return SimpleNamespace(nanoseconds=self.nanoseconds-other.nanoseconds)


class Clock:
    def __init__(self):
        self.seconds = 10.

    def now(self):
        return Time(self.seconds)
